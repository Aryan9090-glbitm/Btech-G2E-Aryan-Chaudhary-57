#include <stdio.h>

int main() {
    long long n;
    int count = 0;

    printf("Enter an integer: ");
    scanf("%lld", &n);

    // Handle the special case where the input is 0
    if (n == 0) {
        count = 1;
    } else {
        // Handle negative numbers by making them positive
        if (n < 0) {
            n = -n;
        }

        // Repeatedly divide by 10 until the number is 0
        while (n != 0) {
            n /= 10;
            count++;
        }
    }

    printf("Number of digits: %d\n", count);

    return 0;
}
