#include <stdio.h>

int main() {
    int i, n, sumEven = 0, sumOdd = 0;

    // Prompt user for the limit N
    printf("Enter the value of N: ");
    scanf("%d", &n);

    // Loop from 1 to N
    for(i = 1; i <= n; i++) {
        if(i % 2 == 0) {
            // Number is even if remainder is 0
            sumEven = sumEven + i;
        } else {
            // Otherwise, it is odd
            sumOdd = sumOdd + i;
        }
    }

    // Output the results
    printf("Sum of all even numbers up to %d = %d\n", n, sumEven);
    printf("Sum of all odd numbers up to %d = %d\n", n, sumOdd);

    return 0;
}
