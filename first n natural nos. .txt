#include <stdio.h>

int main() {
    int i, n;

    // Input the upper limit from user
    printf("Enter the value of n: ");
    scanf("%d", &n);

    printf("The first %d natural numbers are:\n", n);

    /* 
     * Start loop counter from 1 (i=1) and go until n (i<=n)
     * increment the loop count by 1 to get the next value
     */
    for(i = 1; i <= n; i++) {
        printf("%d ", i);
    }

    return 0;
}
