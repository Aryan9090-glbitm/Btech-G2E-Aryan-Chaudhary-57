#include <stdio.h>

int main() {
    int num, i;

    printf("Enter a positive integer: ");
    scanf("%d", &num);

    printf("Factors of %d are: ", num);
    
    // Loop from 1 up to the number itself
    for (i = 1; i <= num; ++i) {
        // If the number is divisible by i, it's a factor
        if (num % i == 0) {
            printf("%d ", i);
        }
    }

    return 0;
}
